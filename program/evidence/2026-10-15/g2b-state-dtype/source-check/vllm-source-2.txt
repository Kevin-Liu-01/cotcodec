    93	def _resolve_gdn_prefill_backend(
    94	    vllm_config: VllmConfig,
    95	) -> tuple[str, Literal["triton", "flashinfer", "cutedsl"]]:
    96	    """Resolve GDN prefill backend.
    97	
    98	    FlashInfer's GDN prefill kernel is chosen when:
    99	    * ``requested in ["flashinfer", "auto"]``;
   100	    * ``platform == cuda``;
   101	    * one of the following:
   102	      - Hopper (SM90) - no further constraints;
   103	      - Blackwell (SM10.x) with ``head_k_dim == 128``, ``cuda_runtime >= 13``;
   104	      - Blackwell (SM12.x) with ``head_k_dim == 128``, ``cuda_runtime >= 13``.
   105	
   106	    In-tree CuteDSL GDN prefill kernel is chosen when:
   107	    * "cutedsl" is requested; (opt-in only)
   108	    * Blackwell (SM10.x) with ``head_k_dim == 128``;
   109	    """
   110	    additional_config = vllm_config.additional_config
   111	    backend_cfg = (
   112	        additional_config.get("gdn_prefill_backend", "auto")
   113	        if isinstance(additional_config, dict)
   114	        else "auto"
   115	    )
   116	    backend = str(backend_cfg).strip().lower()
   117	
   118	    if not current_platform.is_cuda():
   119	        return backend, "triton"
   120	
   121	    head_k_dim = getattr(
   122	        vllm_config.model_config.hf_text_config, "linear_key_head_dim", None
   123	    )
   124	
   125	    supports_flashinfer = False
   126	    supports_cutedsl = False
   127	
   128	    if current_platform.is_device_capability(90):
   129	        supports_flashinfer = True
   130	    elif (
   131	        current_platform.is_device_capability_family(100)
   132	        and head_k_dim == 128
   133	        and current_platform.get_cuda_runtime_major() >= 13
   134	    ):
   135	        supports_flashinfer = True
   136	        supports_cutedsl = True
   137	    elif (
   138	        current_platform.is_device_capability_family(120)
   139	        and head_k_dim == 128
   140	        and current_platform.get_cuda_runtime_major() >= 13
   141	    ):
   142	        # The in-tree CuteDSL kernel targets SM100 only, so it stays off here.
   143	        supports_flashinfer = True
   144	
   145	    if backend in ["flashinfer", "auto"] and supports_flashinfer:
   146	        return backend, "flashinfer"
   147	    if backend == "cutedsl" and supports_cutedsl:
   148	        return backend, "cutedsl"
   149	    return backend, "triton"

   614	    def maybe_disable_tp(self, quant_config: QuantizationConfig | None) -> bool:
   615	        """Whether to replicate ba_proj instead of TP-sharding it.
   616	
   617	        Marlin requires output_size_per_partition >= MIN_THREAD_N=64, which
   618	        the Qwen3.5 non-interleaved [num_v_heads]*2 layout violates at TP>=2
   619	        (e.g. num_v_heads=64, TP=4 -> 16). Replicating the projection keeps
   620	        each rank above the Marlin threshold; forward() then slices b/a to
   621	        the local TP partition. Qwen3-Next's interleaved [num_v_heads*2]
   622	        layout is unaffected and stays TP-sharded.
   623	
   624	        See https://github.com/vllm-project/vllm/issues/35924
   625	        """
   626	        return (
   627	            current_platform.is_cuda()
   628	            and not self.gqa_interleaved_layout
   629	            and isinstance(quant_config, (AutoAWQConfig, AutoGPTQConfig, INCConfig))
   630	        )

   633	        b, a = ba.chunk(2, dim=-1)
   634	        if self.disable_tp_for_ba_proj and self.tp_size > 1:
   635	            # ba_proj is replicated for Marlin; slice b/a to local TP rank.
   636	            ba_chunk = self.num_v_heads // self.tp_size
   637	            ba_start = self.tp_rank * ba_chunk

   639	            a = a[:, ba_start : ba_start + ba_chunk]

   642	    def fix_query_key_value_ordering(
   643	        self,
   644	        mixed_qkvz: torch.Tensor,
   645	        mixed_ba: torch.Tensor,
   646	    ):
   647	        """Derives `query`, `key` and `value` tensors from `mixed_qkvzba`."""
   648	        new_tensor_shape_qkvz = mixed_qkvz.size()[:-1] + (
   649	            self.num_k_heads // self.tp_size,
   650	            (
   651	                self.head_k_dim
   652	                + self.head_k_dim
   653	                + (self.head_v_dim + self.head_v_dim)
   654	                * self.num_v_heads
   655	                // self.num_k_heads
   656	            ),
   657	        )
   658	        new_tensor_shape_ba = mixed_ba.size()[:-1] + (
   659	            self.num_k_heads // self.tp_size,
   660	            2 * self.num_v_heads // self.num_k_heads,
   661	        )
   662	
   663	        mixed_qkvz = mixed_qkvz.view(*new_tensor_shape_qkvz)
   664	        mixed_ba = mixed_ba.view(*new_tensor_shape_ba)
   665	
   666	        split_arg_list_qkvz = [
   667	            self.head_k_dim,
   668	            self.head_k_dim,
   669	            (self.num_v_heads // self.num_k_heads * self.head_v_dim),
   670	            (self.num_v_heads // self.num_k_heads * self.head_v_dim),
   671	        ]
   672	        split_arg_list_ba = [
   673	            self.num_v_heads // self.num_k_heads,
   674	            self.num_v_heads // self.num_k_heads,
   675	        ]
   676	
   677	        # [b, sq, ng, (hn + hn + np/ng * hn + np/ng + np/ng)]
   678	        # --> [b, sq, ng, hn], [b, sq, ng, hn], [b, sq, ng, np/ng * hn],
   679	        #  [b, sq, ng, np/ng * hn], [b, sq, ng, np/ng], [b, sq, ng, np/ng]
   680	        (query, key, value, z) = torch.split(mixed_qkvz, split_arg_list_qkvz, dim=2)
   681	        (b, a) = torch.split(mixed_ba, split_arg_list_ba, dim=2)
   682	
   683	        # [b, sq, ng, np/ng * hn] -> [b, sq, np, hn]
   684	        value = value.reshape(value.size(0), -1, self.head_v_dim)
   685	        z = z.reshape(z.size(0), -1, self.head_v_dim)
   686	        b = b.reshape(b.size(0), self.num_v_heads // self.tp_size)
   687	        a = a.reshape(a.size(0), self.num_v_heads // self.tp_size)
   688	
   689	        return query, key, value, z, b, a
   690	
   691	    @torch.compile(fullgraph=True)
   692	    def prepare_gdn_attention_core_inputs(
   693	        self,
   694	        mixed_qkvz: torch.Tensor,
   695	        mixed_ba: torch.Tensor,
   696	        num_tokens: int,
   697	    ):
   698	        """Derives mixed_qkv, z, b, a from projected qkvz/ba for the GDN custom op.
   699	
   700	        For gqa_interleaved_layout (Qwen3-Next): unpack the interleaved
   701	        [ng, (hk + hk + np/ng*hv + np/ng*hv)] layout into contiguous qkv.
   702	        For non-interleaved layout (Qwen3.5): simple split along last dim.
   703	        """
   704	        if not self.gqa_interleaved_layout:
   705	            # Qwen3.5: weights are in [q, k, v, z] order
   706	            assert num_tokens == mixed_qkvz.shape[0]
   707	            qkv_size = (self.key_dim * 2 + self.value_dim) // self.tp_size
   708	            z_size = self.value_dim // self.tp_size
   709	            mixed_qkv, z_flat = mixed_qkvz.split([qkv_size, z_size], dim=-1)
   710	            n = mixed_qkvz.shape[0]
   711	            z_out = z_flat.reshape(n, -1, self.head_v_dim)
   712	            b, a = mixed_ba.chunk(2, dim=-1)
   713	            return mixed_qkv, z_out, b, a
   714	
   715	        # Qwen3-Next: interleaved GQA layout
   716	        base_shape_qkvz = mixed_qkvz.size()[:-1]
   717	        base_shape_ba = mixed_ba.size()[:-1]
   718	        ng = self.num_k_heads // self.tp_size
   719	
   720	        new_tensor_shape_qkvz = base_shape_qkvz + (
   721	            ng,
   722	            (
   723	                self.head_k_dim
   724	                + self.head_k_dim
   725	                + (self.head_v_dim + self.head_v_dim)
   726	                * self.num_v_heads
   727	                // self.num_k_heads
   728	            ),
   729	        )
   730	        new_tensor_shape_ba = base_shape_ba + (
   731	            ng,
   732	            2 * self.num_v_heads // self.num_k_heads,
   733	        )
   734	
   735	        mixed_qkvz = mixed_qkvz.view(*new_tensor_shape_qkvz)
   736	        mixed_ba = mixed_ba.view(*new_tensor_shape_ba)
   737	
   738	        split_arg_list_qkvz = [
   739	            self.head_k_dim,
   740	            self.head_k_dim,
   741	            (self.num_v_heads // self.num_k_heads * self.head_v_dim),
   742	            (self.num_v_heads // self.num_k_heads * self.head_v_dim),
   743	        ]
   744	        split_arg_list_ba = [
   745	            self.num_v_heads // self.num_k_heads,
   746	            self.num_v_heads // self.num_k_heads,
   747	        ]
   748	
   749	        (query, key, value, z) = torch.split(mixed_qkvz, split_arg_list_qkvz, dim=-1)
   750	        (b, a) = torch.split(mixed_ba, split_arg_list_ba, dim=-1)
   751	
   752	        mixed_qkv_logical = torch.cat(
   753	            [
   754	                query.reshape(num_tokens, -1),
   755	                key.reshape(num_tokens, -1),
   756	                value.reshape(num_tokens, -1),
   757	            ],
   758	            dim=-1,
   759	        )
   760	
   761	        # The split above produces non-contiguous views into the interleaved
   762	        # buffer.  Concatenating everything into a single flat tensor forces a
   763	        # contiguous copy, then slicing back out gives contiguous q/k/v/z/b/a
   764	        # tensors that downstream kernels require.  Doing this in one cat+slice
   765	        # keeps torch.compile in a single Triton graph instead of emitting
   766	        # separate copy kernels per tensor.  The original code used
   767	        # rearrange(...).contiguous() on each tensor individually.
   768	        fused = torch.cat(
   769	            [
   770	                mixed_qkv_logical.reshape(-1),
   771	                z.reshape(-1),
   772	                b.reshape(-1),
   773	                a.reshape(-1),
   774	            ],
   775	            dim=0,
   776	        )
   777	
   778	        curr = 0
   779	        qkv_numel = mixed_qkv_logical.numel()
   780	        z_numel = z.numel()
   781	        b_numel = b.numel()
   782	        a_numel = a.numel()
   783	
   784	        mixed_qkv_out = fused[curr : curr + qkv_numel].view(num_tokens, -1)
   785	        curr += qkv_numel
   786	
   787	        z_out = fused[curr : curr + z_numel].view(
   788	            num_tokens, self.num_v_heads // self.tp_size, self.head_v_dim
   789	        )
   790	        curr += z_numel
   791	
   792	        b_out = fused[curr : curr + b_numel].view(
   793	            num_tokens, self.num_v_heads // self.tp_size
   794	        )
   795	        curr += b_numel
   796	
   797	        a_out = fused[curr : curr + a_numel].view(
   798	            num_tokens, self.num_v_heads // self.tp_size
   799	        )
   800	
   801	        return mixed_qkv_out, z_out, b_out, a_out
   802	
   803	    def rearrange_mixed_qkv(self, mixed_qkv):
   804	        """Split packed qkv into contiguous (1, seq, heads, dim) tensors.
   805	
   806	        The original code used ``rearrange(x, "l (h d) -> 1 l h d", d=...)``
   807	        followed by ``.contiguous()`` on each tensor.  This version flattens
   808	        all three splits into a single buffer via ``torch.cat`` so that
   809	        torch.compile emits one Triton copy kernel instead of three separate
   810	        contiguous() calls.
   811	        """
   812	        if mixed_qkv is None:
   813	            return None, None, None
   814	
   815	        seq_len = mixed_qkv.shape[0]
   816	        q_dim = self.key_dim // self.tp_size
   817	        k_dim = self.key_dim // self.tp_size
   818	        v_dim = self.value_dim // self.tp_size
   819	
   820	        query, key, value = torch.split(mixed_qkv, [q_dim, k_dim, v_dim], dim=-1)
   821	
   822	        fused = torch.cat(
   823	            [query.reshape(-1), key.reshape(-1), value.reshape(-1)], dim=0
   824	        )
   825	
   826	        q_size = seq_len * q_dim
   827	        k_size = seq_len * k_dim
   828	
   829	        q_contig = fused[0:q_size]
   830	        k_contig = fused[q_size : q_size + k_size]
   831	        v_contig = fused[q_size + k_size :]
   832	
   833	        query = q_contig.view(1, seq_len, -1, self.head_k_dim)
   834	        key = k_contig.view(1, seq_len, -1, self.head_k_dim)
   835	        value = v_contig.view(1, seq_len, -1, self.head_v_dim)
   836	
   837	        return query, key, value

   845	    def _output_projection(
   846	        self,
   847	        core_attn_out: torch.Tensor,
   848	        z: torch.Tensor,
   849	    ) -> torch.Tensor:
   850	        """Part 3: RMSNormGated + output linear projection.
   851	
   852	        The RMSNormGated + quant sequence is eligible for fusion
   853	        by the compilation pass when fuse_norm_quant is enabled.
   854	        """
   855	        core_attn_out = self.norm(core_attn_out, z)
   856	        output, _ = self.out_proj(core_attn_out.flatten(-2))
   857	        return output

   859	    def forward_hip(
   860	        self,
   861	        hidden_states: torch.Tensor,
   862	    ) -> torch.Tensor:
   863	        """ROCm forward using AITER Triton fused projection+attention when
   864	        available, otherwise falling back to the generic CUDA path."""
   865	        if GDN_AITER_TRITON_AVAILABLE:
   866	            num_tokens = hidden_states.size(0)
   867	            projected_states_qkvz, _ = self.in_proj_qkvz(hidden_states)
   868	            projected_states_ba, _ = self.in_proj_ba(hidden_states)
   869	            projected_states_qkvz = projected_states_qkvz.view(num_tokens, -1)
   870	            projected_states_ba = projected_states_ba.view(num_tokens, -1)
   871	            core_attn_out = torch.empty(
   872	                (num_tokens, self.num_v_heads // self.tp_size, self.head_v_dim),
   873	                dtype=hidden_states.dtype,
   874	                device=hidden_states.device,
   875	            )
   876	            z = torch.empty(
   877	                (num_tokens, self.num_v_heads // self.tp_size, self.head_v_dim),
   878	                dtype=projected_states_qkvz.dtype,
   879	                device=projected_states_qkvz.device,
   880	            )
   881	
   882	            torch.ops.vllm.qwen_gdn_attention_core(
   883	                projected_states_qkvz,
   884	                projected_states_ba,
   885	                z,
   886	                core_attn_out,
   887	                layer_name=_encode_layer_name(self.prefix),
   888	                use_aiter=True,
   889	            )
   890	
   891	            return self._output_projection(core_attn_out, z)
   892	        else:
   893	            return self.forward_cuda(hidden_states)

   895	    def forward_cuda(
   896	        self,
   897	        hidden_states: torch.Tensor,
   898	    ) -> torch.Tensor:
   899	        """Forward pass with three parts:
   900	        1. Input projection
   901	        2. Core attention (custom op)
   902	        3. Output projection
   903	        """
   904	        num_tokens = hidden_states.size(0)
   905	        # ============================================================
   906	        # Part 1: Input Projection
   907	        # ============================================================
   908	        mixed_qkvz, _ = self.in_proj_qkvz(hidden_states)
   909	        ba, _ = self.in_proj_ba(hidden_states)
   910	
   911	        use_fused_gdn_decode = (
   912	            self.enable_fused_gdn_decode
   913	            and hidden_states.dtype == torch.bfloat16
   914	            and self.norm.weight.dtype in (torch.bfloat16, torch.float32)
   915	        )
   916	        if use_fused_gdn_decode:
   917	            core_attn_out = torch.zeros(
   918	                (num_tokens, self.num_v_heads // self.tp_size, self.head_v_dim),
   919	                dtype=hidden_states.dtype,
   920	                device=hidden_states.device,
   921	            )
   922	            torch.ops.vllm.qwen_gdn_attention_core_fused_norm_packed(
   923	                mixed_qkvz,
   924	                ba,
   925	                core_attn_out,
   926	                layer_name=_encode_layer_name(self.prefix),
   927	            )
   928	            output, _ = self.out_proj(core_attn_out.flatten(-2))
   929	            return output
   930	
   931	        if self.gqa_interleaved_layout:
   932	            # Qwen3-Next: unpack the interleaved GQA layout
   933	            query, key, value, z, b, a = self.fix_query_key_value_ordering(
   934	                mixed_qkvz, ba
   935	            )
   936	            query, key, value = map(
   937	                lambda x: rearrange(x, "l p d -> l (p d)"), (query, key, value)
   938	            )
   939	            mixed_qkv = torch.cat((query, key, value), dim=-1)
   940	        else:
   941	            # Qwen3.5: weights are already in [q, k, v, z] and [b, a] order
   942	            qkv_size = (self.key_dim * 2 + self.value_dim) // self.tp_size
   943	            z_size = self.value_dim // self.tp_size
   944	            mixed_qkv, z = mixed_qkvz.split([qkv_size, z_size], dim=-1)
   945	            z = z.reshape(z.size(0), -1, self.head_v_dim)
   946	            b, a = self.split_ba(ba)
   947	
   948	        # ============================================================
   949	        # Part 2: Core Attention (Custom Op)
   950	        # ============================================================
   951	        # Note: we should not use torch.empty here like other attention backends,
   952	        # see discussions in https://github.com/vllm-project/vllm/pull/28182
   953	        core_attn_out = torch.zeros(
   954	            (num_tokens, self.num_v_heads // self.tp_size, self.head_v_dim),
   955	            dtype=hidden_states.dtype,
   956	            device=hidden_states.device,
   957	        )
   958	
   959	        torch.ops.vllm.qwen_gdn_attention_core(
   960	            mixed_qkv,
   961	            b.contiguous(),
   962	            a.contiguous(),
   963	            core_attn_out,
   964	            layer_name=_encode_layer_name(self.prefix),
   965	        )
   966	
   967	        # ============================================================
   968	        # Part 3: Output Projection
   969	        # ============================================================
   970	        return self._output_projection(core_attn_out, z)
   971

   973	        self,
   974	        hidden_states: torch.Tensor,
   975	    ) -> torch.Tensor:
   976	        """Forward pass with three parts:
   977	        1. Input projection

   979	        3. Output projection

   981	        num_tokens = hidden_states.size(0)

   983	        # ============================================================

  1055	    def _warmup_prefill_kernels(self, qkv_or_qkvz: torch.Tensor, v_dim: int) -> None:
  1056	        """Warm up GDN prefill kernels during V1 profiling.
  1057	
  1058	        During V1 profile runs, ``_forward_core`` returns early because
  1059	        ``attn_metadata`` is ``None``, so the autotuned kernels used by
  1060	        ``chunk_gated_delta_rule`` (e.g. ``solve_tril``,
  1061	        ``chunk_scaled_dot_kkt``) are never invoked.  After profiling,
  1062	        vLLM allocates KV cache using most of the remaining GPU memory.
  1063	        When the first real inference triggers the autotuner it OOMs
  1064	        because there is not enough memory left for benchmarking.
  1065	
  1066	        This method runs minimal forward passes through
  1067	        ``chunk_gated_delta_rule`` with small dummy tensors to force
  1068	        autotuning while GPU memory is still plentiful.  The autotuner
  1069	        results are cached globally, so only the first layer incurs
  1070	        actual benchmarking cost.
  1071	
  1072	        All kernels including ``chunk_fwd_kernel_o`` now use a fixed
  1073	        ``BT = chunk_size`` (64).  A single warmup pass with T = 64
  1074	        is sufficient to populate the autotuner cache.
  1075	
  1076	        The decode path uses ``gdn_aiter_fused_rearrange_sigmoid_gated_delta_rule``
  1077	        which has fixed kernel parameters (no autotuning), so only the
  1078	        prefill (chunked) path needs warming up.
  1079	        """
  1080	        if self._prefill_kernels_warmed_up:
  1081	            return
  1082	        self._prefill_kernels_warmed_up = True
  1083	
  1084	        device = qkv_or_qkvz.device
  1085	        dtype = qkv_or_qkvz.dtype
  1086	        num_k_heads = self.num_k_heads // self.tp_size
  1087	        num_v_heads = self.num_v_heads // self.tp_size
  1088	        _, state_dtype = self.get_state_dtype()
  1089	
  1090	        # All kernels use BT = chunk_size, so a single pass with T = chunk_size
  1091	        # is sufficient to populate every autotuner cache. Mirror the real
  1092	        # prefill path here: build q/k/v/g/beta via fused_post_conv_prep and
  1093	        # then run chunk_gated_delta_rule with in-kernel L2 norm disabled.
  1094	        T = FLA_CHUNK_SIZE
  1095	        dummy_mixed_qkv = torch.randn(
  1096	            T, qkv_or_qkvz.shape[-1] - v_dim, device=device, dtype=dtype
  1097	        )
  1098	        dummy_a = torch.randn(T, num_v_heads, device=device, dtype=dtype)
  1099	        dummy_b = torch.randn(T, num_v_heads, device=device, dtype=dtype)
  1100	        q, k, v, g, beta = fused_post_conv_prep(
  1101	            conv_output=dummy_mixed_qkv,
  1102	            a=dummy_a,
  1103	            b=dummy_b,
  1104	            A_log=self.A_log,
  1105	            dt_bias=self.dt_bias,
  1106	            num_k_heads=num_k_heads,
  1107	            head_k_dim=self.head_k_dim,
  1108	            head_v_dim=self.head_v_dim,
  1109	            apply_l2norm=True,
  1110	            output_g_exp=False,
  1111	        )
  1112	        q = q.unsqueeze(0)
  1113	        k = k.unsqueeze(0)
  1114	        v = v.unsqueeze(0)
  1115	        g = g.unsqueeze(0)
  1116	        beta = beta.unsqueeze(0)
  1117	        state = torch.zeros(
  1118	            1,
  1119	            num_v_heads,
  1120	            self.head_v_dim,
  1121	            self.head_k_dim,
  1122	            device=device,
  1123	            dtype=state_dtype,
  1124	        )
  1125	        cu_seqlens = torch.tensor([0, T], device=device, dtype=torch.int32)
  1126	
  1127	        # CuteDSL kernels require metadata
  1128	        chunk_indices = None
  1129	        chunk_offsets = None
  1130	        if self.gdn_prefill_backend == "cutedsl":
  1131	            from vllm.model_executor.layers.mamba.ops.gdn_chunk_cutedsl import (
  1132	                prepare_metadata_cutedsl,
  1133	            )
  1134	
  1135	            chunk_indices, chunk_offsets = prepare_metadata_cutedsl(cu_seqlens, T)
  1136	
  1137	        try:
  1138	            self.chunk_gated_delta_rule(
  1139	                q=q,
  1140	                k=k,
  1141	                v=v,
  1142	                g=g,
  1143	                beta=beta,
  1144	                initial_state=state,
  1145	                output_final_state=True,
  1146	                cu_seqlens=cu_seqlens,
  1147	                chunk_indices=chunk_indices,
  1148	                chunk_offsets=chunk_offsets,
  1149	                use_qk_l2norm_in_kernel=False,
  1150	            )
  1151	        except Exception:
  1152	            logger.warning(
  1153	                "GDN prefill kernel warmup (T=%d) failed for "
  1154	                "layer %s. First inference may OOM due to "
  1155	                "autotuner.",
  1156	                T,
  1157	                self.prefix,
  1158	                exc_info=True,
  1159	            )
  1160	        else:
  1161	            logger.debug(
  1162	                "GDN prefill kernel warmup (T=%d) completed for layer %s",
  1163	                T,
  1164	                self.prefix,
  1165	            )
  1166	        finally:
  1167	            del (
  1168	                dummy_mixed_qkv,
  1169	                q,
  1170	                k,
  1171	                v,

  1184	    def _forward_core_rocm(
  1185	        self,

  1188	        z_out: torch.Tensor,
  1189	        core_attn_out: torch.Tensor,
  1190	    ):
  1191	        """ROCm AITER fast path: conv1d + recurrent attention from packed
  1192	        qkvz/ba layout.
  1193	
  1194	        For decode-only (no spec, no prefill) batches, dispatches directly to
  1195	        ``_forward_core_decode_aiter``. Otherwise unpacks the packed layout and
  1196	        falls through to ``_forward_core``.
  1197	
  1198	        Args:
  1199	            qkvz: packed [q, k, v, z] projection (num_tokens, qkvz_dim)

  1202	                   head_dim); mutated in-place.
  1203	            core_attn_out: Pre-allocated output buffer for attention results.
  1204	
  1205	        """
  1206	        forward_context = get_forward_context()
  1207	        attn_metadata_raw = forward_context.attn_metadata
  1208	
  1209	        attn_metadata = None
  1210	        if isinstance(attn_metadata_raw, dict):
  1211	            attn_metadata = attn_metadata_raw.get(self.prefix)
  1212	        if attn_metadata is None:
  1213	            v_dim = core_attn_out.shape[-1] * core_attn_out.shape[-2]
  1214	            self._warmup_prefill_kernels(qkvz, v_dim)
  1215	            return
  1216	
  1217	        assert isinstance(attn_metadata, GDNAttentionMetadata)
  1218	
  1219	        if (
  1220	            attn_metadata.spec_sequence_masks is None
  1221	            and attn_metadata.num_prefills == 0
  1222	            and attn_metadata.num_decodes > 0
  1223	        ):
  1224	            return self._forward_core_decode_aiter(
  1225	                qkvz=qkvz,
  1226	                ba=ba,
  1227	                z_out=z_out,
  1228	                core_attn_out=core_attn_out,
  1229	                attn_metadata=attn_metadata,
  1230	            )
  1231	
  1232	        core_attn_out.zero_()
  1233	        num_tokens_all = qkvz.shape[0]
  1234	        mixed_qkv, z, b, a = self.prepare_gdn_attention_core_inputs(
  1235	            qkvz, ba, num_tokens_all
  1236	        )
  1237	        z_out[:] = z
  1238	        self._forward_core(
  1239	            mixed_qkv=mixed_qkv,
  1240	            b=b,
  1241	            a=a,
  1242	            core_attn_out=core_attn_out,
  1243	        )

  1245	    def _forward_core(
  1246	        self,
  1247	        mixed_qkv: torch.Tensor,
  1248	        b: torch.Tensor,
  1249	        a: torch.Tensor,
  1250	        core_attn_out: torch.Tensor,
  1251	    ):
  1252	        """Core conv1d + recurrent attention (standard path).
  1253	
  1254	        Args:
  1255	            mixed_qkv: packed [q, k, v] projection (num_tokens, qkv_dim)
  1256	            b: beta gating vector                   (num_tokens, num_heads)
  1257	            a: alpha gating vector                  (num_tokens, num_heads)
  1258	            core_attn_out: Pre-allocated output buffer for attention results.
  1259	
  1260	        """
  1261	        forward_context = get_forward_context()
  1262	        attn_metadata_raw = forward_context.attn_metadata
  1263	
  1264	        attn_metadata = None
  1265	        if isinstance(attn_metadata_raw, dict):
  1266	            attn_metadata = attn_metadata_raw.get(self.prefix)
  1267	        if attn_metadata is None:
  1268	            self._warmup_prefill_kernels(mixed_qkv, 0)
  1269	            return
  1270	
  1271	        assert isinstance(attn_metadata, GDNAttentionMetadata)
  1272	
  1273	        if (
  1274	            self.enable_packed_recurrent_decode
  1275	            and attn_metadata.spec_sequence_masks is None
  1276	            and attn_metadata.num_prefills == 0
  1277	            and attn_metadata.num_decodes > 0
  1278	        ):
  1279	            return self._forward_core_decode_non_spec(
  1280	                mixed_qkv=mixed_qkv,
  1281	                b=b,
  1282	                a=a,
  1283	                core_attn_out=core_attn_out,
  1284	                attn_metadata=attn_metadata,
  1285	            )
  1286	
  1287	        has_initial_state = attn_metadata.has_initial_state
  1288	        spec_query_start_loc = attn_metadata.spec_query_start_loc
  1289	        non_spec_query_start_loc = attn_metadata.non_spec_query_start_loc
  1290	        spec_sequence_masks = attn_metadata.spec_sequence_masks
  1291	        spec_token_indx = attn_metadata.spec_token_indx
  1292	        non_spec_token_indx = attn_metadata.non_spec_token_indx
  1293	        spec_state_indices_tensor = attn_metadata.spec_state_indices_tensor  # noqa: E501
  1294	        non_spec_state_indices_tensor = attn_metadata.non_spec_state_indices_tensor  # noqa: E501
  1295	        self_kv_cache = self.kv_cache
  1296	        # conv_state must be (..., dim, width-1) for the conv kernels.
  1297	        # DS layout stores it that way directly; SD layout needs a transpose.
  1298	        conv_state = (
  1299	            self_kv_cache[0]
  1300	            if is_conv_state_dim_first()
  1301	            else self_kv_cache[0].transpose(-1, -2)
  1302	        )
  1303	        ssm_state = self_kv_cache[1]
  1304	        num_actual_tokens = attn_metadata.num_actual_tokens
  1305	        num_accepted_tokens = attn_metadata.num_accepted_tokens
  1306	
  1307	        mixed_qkv = mixed_qkv[:num_actual_tokens]
  1308	        b = b[:num_actual_tokens]
  1309	        a = a[:num_actual_tokens]
  1310	
  1311	        # 1. Convolution sequence transformation
  1312	        conv_weights = self.conv1d.weight.view(
  1313	            self.conv1d.weight.size(0), self.conv1d.weight.size(2)
  1314	        )
  1315	
  1316	        if spec_sequence_masks is not None:
  1317	            if attn_metadata.num_prefills == 0 and attn_metadata.num_decodes == 0:
  1318	                mixed_qkv_spec = mixed_qkv
  1319	                a_spec = a
  1320	                b_spec = b
  1321	                mixed_qkv_non_spec = None
  1322	            else:
  1323	                mixed_qkv_spec = mixed_qkv.index_select(0, spec_token_indx)
  1324	                a_spec = a.index_select(0, spec_token_indx)
  1325	                b_spec = b.index_select(0, spec_token_indx)
  1326	                mixed_qkv_non_spec = mixed_qkv.index_select(0, non_spec_token_indx)
  1327	        else:
  1328	            mixed_qkv_spec = None
  1329	            mixed_qkv_non_spec = mixed_qkv
  1330	
  1331	        # 1.1: Process the multi-query part
  1332	        if spec_sequence_masks is not None:
  1333	            # spec_state_indices_tensor is always set when spec_sequence_masks is set
  1334	            assert spec_state_indices_tensor is not None
  1335	            mixed_qkv_spec = causal_conv1d_update(
  1336	                mixed_qkv_spec,
  1337	                conv_state,
  1338	                conv_weights,
  1339	                self.conv1d.bias,
  1340	                self.activation,
  1341	                conv_state_indices=spec_state_indices_tensor[:, 0][  # type: ignore[index]
  1342	                    : attn_metadata.num_spec_decodes  # type: ignore[attr-defined]
  1343	                ],
  1344	                num_accepted_tokens=num_accepted_tokens,
  1345	                query_start_loc=spec_query_start_loc,
  1346	                max_query_len=spec_state_indices_tensor.size(-1),
  1347	                validate_data=False,
  1348	            )
  1349	
  1350	        # 1.2: Process the remaining part
  1351	        if attn_metadata.num_prefills > 0:
  1352	            assert mixed_qkv_non_spec is not None
  1353	            mixed_qkv_non_spec_T = mixed_qkv_non_spec.transpose(0, 1)
  1354	            # - "cache_indices" updates the conv_state cache in positions
  1355	            #   pointed to by "state_indices_tensor"
  1356	            mixed_qkv_non_spec = causal_conv1d_fn(
  1357	                mixed_qkv_non_spec_T,
  1358	                conv_weights,
  1359	                self.conv1d.bias,
  1360	                activation=self.activation,
  1361	                conv_states=conv_state,
  1362	                has_initial_state=has_initial_state,
  1363	                cache_indices=non_spec_state_indices_tensor,
  1364	                query_start_loc=non_spec_query_start_loc,
  1365	                metadata=attn_metadata,
  1366	            ).transpose(0, 1)
  1367	        elif attn_metadata.num_decodes > 0:
  1368	            assert mixed_qkv_non_spec is not None
  1369	            mixed_qkv_non_spec = causal_conv1d_update(
  1370	                mixed_qkv_non_spec,
  1371	                conv_state,
  1372	                conv_weights,
  1373	                self.conv1d.bias,
  1374	                self.activation,
  1375	                conv_state_indices=non_spec_state_indices_tensor[  # type: ignore[index]
  1376	                    : attn_metadata.num_actual_tokens  # type: ignore[attr-defined]
  1377	                ],
  1378	                validate_data=True,
  1379	            )
  1380	        else:
  1381	            mixed_qkv_non_spec = None
  1382	
  1383	        query_spec, key_spec, value_spec = self.rearrange_mixed_qkv(mixed_qkv_spec)
  1384	
  1385	        # Split mixed non-spec-decode+prefill to process independently
  1386	        split_non_spec = (
  1387	            spec_sequence_masks is None
  1388	            and attn_metadata.num_prefills > 0
  1389	            and attn_metadata.num_decodes > 0
  1390	        )
  1391	        num_decode_tokens = attn_metadata.num_decode_tokens
  1392	
  1393	        if attn_metadata.num_prefills > 0:
  1394	            assert mixed_qkv_non_spec is not None, (
  1395	                "mixed_qkv_non_spec must be provided for prefill path"
  1396	            )
  1397	            if spec_sequence_masks is not None:
  1398	                a_non_spec = a.index_select(0, non_spec_token_indx)
  1399	                b_non_spec = b.index_select(0, non_spec_token_indx)
  1400	            else:
  1401	                a_non_spec = a
  1402	                b_non_spec = b
  1403	
  1404	            if split_non_spec:
  1405	                conv_output_prefill = mixed_qkv_non_spec[num_decode_tokens:]
  1406	                a_prefill = a_non_spec[num_decode_tokens:]
  1407	                b_prefill = b_non_spec[num_decode_tokens:]
  1408	            else:
  1409	                conv_output_prefill = mixed_qkv_non_spec
  1410	                a_prefill = a_non_spec
  1411	                b_prefill = b_non_spec
  1412	
  1413	            (
  1414	                query_non_spec,
  1415	                key_non_spec,
  1416	                value_non_spec,
  1417	                g_non_spec,
  1418	                beta_non_spec,
  1419	            ) = fused_post_conv_prep(
  1420	                conv_output=conv_output_prefill,
  1421	                a=a_prefill,
  1422	                b=b_prefill,
  1423	                A_log=self.A_log,
  1424	                dt_bias=self.dt_bias,
  1425	                num_k_heads=self.num_k_heads // self.tp_size,
  1426	                head_k_dim=self.head_k_dim,
  1427	                head_v_dim=self.head_v_dim,
  1428	                apply_l2norm=True,
  1429	                output_g_exp=False,
  1430	            )
  1431	            query_non_spec = query_non_spec.unsqueeze(0)
  1432	            key_non_spec = key_non_spec.unsqueeze(0)
  1433	            value_non_spec = value_non_spec.unsqueeze(0)
  1434	            g_non_spec = g_non_spec.unsqueeze(0)
  1435	            beta_non_spec = beta_non_spec.unsqueeze(0)
  1436	        else:
  1437	            query_non_spec, key_non_spec, value_non_spec = self.rearrange_mixed_qkv(
  1438	                mixed_qkv_non_spec
  1439	            )
  1440	            g_non_spec = None
  1441	            beta_non_spec = None
  1442	
  1443	        # 2. Recurrent attention
  1444	
  1445	        # 2.1: Process the multi-query part
  1446	        if spec_sequence_masks is not None:
  1447	            core_attn_out_spec, last_recurrent_state = (
  1448	                fused_sigmoid_gating_delta_rule_update(
  1449	                    A_log=self.A_log,
  1450	                    a=a_spec,
  1451	                    b=b_spec,
  1452	                    dt_bias=self.dt_bias,
  1453	                    q=query_spec,
  1454	                    k=key_spec,
  1455	                    v=value_spec,
  1456	                    initial_state=ssm_state,
  1457	                    inplace_final_state=True,
  1458	                    cu_seqlens=spec_query_start_loc[  # type: ignore[index]
  1459	                        : attn_metadata.num_spec_decodes
  1460	                        + 1  # type: ignore[attr-defined]
  1461	                    ],
  1462	                    ssm_state_indices=spec_state_indices_tensor,
  1463	                    num_accepted_tokens=num_accepted_tokens,
  1464	                    use_qk_l2norm_in_kernel=True,
  1465	                )
  1466	            )
  1467	        else:
  1468	            core_attn_out_spec, last_recurrent_state = None, None
  1469	
  1470	        # 2.2: Process non-spec-decode part
  1471	        if split_non_spec:
  1472	            query_decode, key_decode, value_decode = self.rearrange_mixed_qkv(
  1473	                mixed_qkv_non_spec[:num_decode_tokens]  # type: ignore[index]
  1474	            )
  1475	            core_attn_out_decode, _ = fused_sigmoid_gating_delta_rule_update(
  1476	                A_log=self.A_log,
  1477	                a=a[:num_decode_tokens],
  1478	                b=b[:num_decode_tokens],
  1479	                dt_bias=self.dt_bias,
  1480	                q=query_decode,
  1481	                k=key_decode,
  1482	                v=value_decode,
  1483	                initial_state=ssm_state,
  1484	                inplace_final_state=True,
  1485	                cu_seqlens=non_spec_query_start_loc[  # type: ignore[index]
  1486	                    : attn_metadata.num_decodes + 1
  1487	                ],
  1488	                ssm_state_indices=non_spec_state_indices_tensor,
  1489	                use_qk_l2norm_in_kernel=True,
  1490	            )
  1491	        else:
  1492	            core_attn_out_decode = None
  1493	
  1494	        # 2.3: Process the remaining part (prefill chunk, or non-spec decode-only)
  1495	        if attn_metadata.num_prefills > 0:
  1496	            # State indices, initial-state mask and cu_seqlens for the chunk
  1497	            # kernel are precomputed by the metadata builder (the prefill tail
  1498	            # when decodes are peeled off, else the full non-spec batch), so they
  1499	            # don't need to be re-derived per layer.
  1500	            prefill_state_indices = attn_metadata.prefill_state_indices
  1501	            prefill_has_initial_state = attn_metadata.prefill_has_initial_state
  1502	            assert prefill_state_indices is not None
  1503	            assert prefill_has_initial_state is not None
  1504	            initial_state = ssm_state[prefill_state_indices]
  1505	            initial_state[~prefill_has_initial_state, ...] = 0
  1506	            (
  1507	                core_attn_out_non_spec,
  1508	                last_recurrent_state,
  1509	            ) = self.chunk_gated_delta_rule(
  1510	                q=query_non_spec,
  1511	                k=key_non_spec,
  1512	                v=value_non_spec,
  1513	                g=g_non_spec,
  1514	                beta=beta_non_spec,
  1515	                initial_state=initial_state,
  1516	                output_final_state=True,
  1517	                cu_seqlens=attn_metadata.prefill_query_start_loc,
  1518	                chunk_indices=attn_metadata.chunk_indices,
  1519	                chunk_offsets=attn_metadata.chunk_offsets,
  1520	                use_qk_l2norm_in_kernel=False,
  1521	            )
  1522	            # Init cache
  1523	            ssm_state[prefill_state_indices] = last_recurrent_state.to(ssm_state.dtype)
  1524	
  1525	            if split_non_spec:
  1526	                # Stitch the peeled decode outputs in front of the prefill
  1527	                # outputs (decode-first order).
  1528	                core_attn_out_non_spec = torch.cat(
  1529	                    [core_attn_out_decode, core_attn_out_non_spec], dim=1
  1530	                )
  1531	        elif attn_metadata.num_decodes > 0:
  1532	            core_attn_out_non_spec, last_recurrent_state = (
  1533	                fused_sigmoid_gating_delta_rule_update(
  1534	                    A_log=self.A_log,
  1535	                    a=a,
  1536	                    b=b,
  1537	                    dt_bias=self.dt_bias,
  1538	                    q=query_non_spec,
  1539	                    k=key_non_spec,
  1540	                    v=value_non_spec,
  1541	                    initial_state=ssm_state,
  1542	                    inplace_final_state=True,
  1543	                    cu_seqlens=non_spec_query_start_loc[  # type: ignore[index]
  1544	                        : attn_metadata.num_decodes
  1545	                        + 1  # type: ignore[attr-defined]
  1546	                    ],
  1547	                    ssm_state_indices=non_spec_state_indices_tensor,
  1548	                    use_qk_l2norm_in_kernel=True,
  1549	                )
  1550	            )
  1551	        else:
  1552	            core_attn_out_non_spec, last_recurrent_state = None, None
  1553	
  1554	        # 3. Merge core attention output
  1555	        if spec_sequence_masks is not None and core_attn_out_non_spec is not None:
  1556	            core_attn_out.index_copy_(0, spec_token_indx, core_attn_out_spec.squeeze(0))
  1557	            core_attn_out.index_copy_(
  1558	                0, non_spec_token_indx, core_attn_out_non_spec.squeeze(0)
  1559	            )
  1560	        elif spec_sequence_masks is not None:
  1561	            core_attn_out[:num_actual_tokens] = core_attn_out_spec.squeeze(0)
  1562	        else:
  1563	            core_attn_out[:num_actual_tokens] = core_attn_out_non_spec.squeeze(0)
